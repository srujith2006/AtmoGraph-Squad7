#include <stdio.h>

int main()
{
    int ps, n, f[10], la;
    int page, offset, pa, i;

    printf("Enter page size: ");
    scanf("%d", &ps);

    printf("Enter number of pages: ");
    scanf("%d", &n);

    printf("Enter frame numbers:\n");

    for(i = 0; i < n; i++)
    {
        scanf("%d", &f[i]);
    }

    printf("Enter logical address: ");
    scanf("%d", &la);

    page = la / ps;
    offset = la % ps;

    if(page >= n)
    {
        printf("Invalid address");
    }
    else
    {
        pa = f[page] * ps + offset;

        printf("Page Number = %d\n", page);
        printf("Offset = %d\n", offset);
        printf("Physical Address = %d\n", pa);
    }

    return 0;
}


FIFO:
#include <stdio.h>

int main()
{
    int n, f, i, j, k = 0;
    int pages[50], frame[10];
    int faults = 0, hits = 0, flag;

    printf("Enter number of pages: ");
    scanf("%d", &n);

    printf("Enter page reference string: ");
    for(i = 0; i < n; i++)
        scanf("%d", &pages[i]);

    printf("Enter number of frames: ");
    scanf("%d", &f);

    for(i = 0; i < f; i++)
        frame[i] = -1;

    printf("\nPage\tFrames\t\tStatus\n");

    for(i = 0; i < n; i++)
    {
        flag = 0;

        for(j = 0; j < f; j++)
        {
            if(frame[j] == pages[i])
            {
                flag = 1;
                hits++;
                break;
            }
        }

        if(flag == 0)
        {
            frame[k] = pages[i];
            k = (k + 1) % f;
            faults++;
        }

        printf("%d\t", pages[i]);

        for(j = 0; j < f; j++)
        {
            if(frame[j] == -1)
                printf("- ");
            else
                printf("%d ", frame[j]);
        }

        if(flag == 0)
            printf("\tPage Fault");
        else
            printf("\tPage Hit");

        printf("\n");
    }

    printf("\nTotal Page Faults = %d", faults);
    printf("\nTotal Page Hits = %d", hits);

    return 0;
}

readers writer:
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>

sem_t mutex, wrt;
int readcount = 0;

void *reader(void *arg)
{
    sem_wait(&mutex);
    readcount++;

    if(readcount == 1)
        sem_wait(&wrt);

    sem_post(&mutex);

    printf("Reader is reading\n");

    sem_wait(&mutex);
    readcount--;

    if(readcount == 0)
        sem_post(&wrt);

    sem_post(&mutex);

    return NULL;
}

void *writer(void *arg)
{
    sem_wait(&wrt);

    printf("Writer is writing\n");

    sem_post(&wrt);

    return NULL;
}

int main()
{
    pthread_t r, w;

    sem_init(&mutex, 0, 1);
    sem_init(&wrt, 0, 1);

    pthread_create(&r, NULL, reader, NULL);
    pthread_create(&w, NULL, writer, NULL);

    pthread_join(r, NULL);
    pthread_join(w, NULL);

    sem_destroy(&mutex);
    sem_destroy(&wrt);

    return 0;
}


producer_consumer:
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>

sem_t empty, full;
int buffer = 0;

void *producer(void *arg)
{
    int i;
    for(i=1; i<=5; i++)
    {
        sem_wait(&empty);
        buffer = i;
        printf("Producer produced: %d\n", buffer);
        sem_post(&full);
    }
    return NULL;
}

void *consumer(void *arg)
{
    int i;
    for(i=1; i<=5; i++)
    {
        sem_wait(&full);
        printf("Consumer consumed: %d\n", buffer);
        sem_post(&empty);
    }
    return NULL;
}

int main()
{
    pthread_t p, c;

    sem_init(&empty, 0, 1);
    sem_init(&full, 0, 0);

    pthread_create(&p, NULL, producer, NULL);
    pthread_create(&c, NULL, consumer, NULL);

    pthread_join(p, NULL);
    pthread_join(c, NULL);

    sem_destroy(&empty);
    sem_destroy(&full);

    return 0;
}

#include <stdio.h>

int main()
{
    int n, i, j, time = 0, min, t;
    int at[10], bt[10], pr[10];
    int ft[10], wt[10], tat[10], p[10];
    int done[10] = {0};
    float avgwt = 0, avgtat = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        p[i] = i + 1;
        printf("Enter AT, BT and Priority of P%d: ", i+1);
        scanf("%d%d%d", &at[i], &bt[i], &pr[i]);
    }

    printf("\nGantt Chart:\n");

    for(i = 0; i < n; i++)
    {
        min = -1;

        for(j = 0; j < n; j++)
        {
            if(!done[j] && at[j] <= time)
            {
                if(min == -1 || pr[j] < pr[min])
                    min = j;
            }
        }

        if(min == -1)
        {
            time++;
            i--;
            continue;
        }

        if(i == 0)
            printf("%d", time);

        time += bt[min];
        ft[min] = time;
        done[min] = 1;

        printf(" | P%d | %d", p[min], time);
    }

    printf("\n\nP\tAT\tBT\tPR\tFT\tWT\tTAT\n");

    for(i = 0; i < n; i++)
    {
        tat[i] = ft[i] - at[i];
        wt[i] = tat[i] - bt[i];

        avgwt += wt[i];
        avgtat += tat[i];

        printf("P%d\t%d\t%d\t%d\t%d\t%d\t%d\n",
        p[i], at[i], bt[i], pr[i], ft[i], wt[i], tat[i]);
    }

    printf("\nAverage WT = %.2f", avgwt/n);
    printf("\nAverage TAT = %.2f", avgtat/n);

    return 0;
}

#include <stdio.h>

int main()
{
    int n, tq, i, time = 0;
    int bt[10], rem[10], wt[10], tat[10];

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("Enter BT of P%d: ", i+1);
        scanf("%d", &bt[i]);
        rem[i] = bt[i];
    }

    printf("Enter time quantum: ");
    scanf("%d", &tq);

    printf("\nGantt Chart:\n");
    printf("0");

    while(1)
    {
        int done = 1;

        for(i = 0; i < n; i++)
        {
            if(rem[i] > 0)
            {
                done = 0;

                if(rem[i] > tq)
                {
                    time = time + tq;
                    rem[i] = rem[i] - tq;
                }
                else
                {
                    time = time + rem[i];
                    rem[i] = 0;
                }

                tat[i] = time;
                printf(" -- P%d -- %d", i+1, time);
            }
        }

        if(done == 1)
            break;
    }

    float awt = 0, atat = 0;

    printf("\n\nProcess\tBT\tWT\tTAT\n");

    for(i = 0; i < n; i++)
    {
        wt[i] = tat[i] - bt[i];

        printf("P%d\t%d\t%d\t%d\n",
               i+1, bt[i], wt[i], tat[i]);

        awt = awt + wt[i];
        atat = atat + tat[i];
    }

    printf("\nAverage WT = %.2f", awt/n);
    printf("\nAverage TAT = %.2f", atat/n);

    return 0;
}

#include <stdio.h>

int main()
{
    int n, i, j, temp, time = 0;
    int bt[20], p[20], wt[20], tat[20];
    float avgwt = 0, avgtat = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        p[i] = i + 1;
        printf("Enter burst time of P%d: ", p[i]);
        scanf("%d", &bt[i]);
    }

    // Sort according to burst time
    for(i = 0; i < n-1; i++)
    {
        for(j = i+1; j < n; j++)
        {
            if(bt[i] > bt[j])
            {
                temp = bt[i];
                bt[i] = bt[j];
                bt[j] = temp;

                temp = p[i];
                p[i] = p[j];
                p[j] = temp;
            }
        }
    }

    wt[0] = 0;

    for(i = 1; i < n; i++)
        wt[i] = wt[i-1] + bt[i-1];

    for(i = 0; i < n; i++)
    {
        tat[i] = wt[i] + bt[i];
        avgwt += wt[i];
        avgtat += tat[i];
    }

    printf("\nProcess\tBT\tWT\tTAT\n");

    for(i = 0; i < n; i++)
    {
        printf("P%d\t%d\t%d\t%d\n",
               p[i], bt[i], wt[i], tat[i]);
    }

    // Gantt Chart
    printf("\nGantt Chart:\n");
    printf("%d", time);

    for(i = 0; i < n; i++)
    {
        time = time + bt[i];
        printf(" -- P%d -- %d", p[i], time);
    }

    printf("\n\nAverage Waiting Time = %.2f", avgwt/n);
    printf("\nAverage Turnaround Time = %.2f", avgtat/n);

    return 0;
}

FCFS:
#include <stdio.h>

int main()
{
    int n, i;
    int at[10], bt[10], ft[10], wt[10], tat[10];
    float avgwt = 0, avgtat = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("Enter AT and BT of P%d: ", i+1);
        scanf("%d%d", &at[i], &bt[i]);
    }

    ft[0] = at[0] + bt[0];

    for(i = 1; i < n; i++)
    {
        if(ft[i-1] < at[i])
            ft[i] = at[i] + bt[i];
        else
            ft[i] = ft[i-1] + bt[i];
    }

    for(i = 0; i < n; i++)
    {
        tat[i] = ft[i] - at[i];
        wt[i] = tat[i] - bt[i];

        avgwt += wt[i];
        avgtat += tat[i];
    }

    printf("\nP\tAT\tBT\tFT\tWT\tTAT\n");

    for(i = 0; i < n; i++)
    {
        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
        i+1, at[i], bt[i], ft[i], wt[i], tat[i]);
    }

    printf("\nGANTT CHART\n");

    printf("%d", at[0]);

    for(i = 0; i < n; i++)
    {
        printf(" | P%d | %d", i+1, ft[i]);
    }

    printf("\n\nAverage WT = %.2f", avgwt/n);
    printf("\nAverage TAT = %.2f", avgtat/n);

    return 0;
}
